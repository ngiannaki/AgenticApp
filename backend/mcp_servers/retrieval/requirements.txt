# Packages the retrieval server needs beyond requirements-base.txt (which
# already pins fastmcp, openai and psycopg — the Postgres/pgvector store is
# backend/core/vector_store/pg_store.py).
#
# --- v2 pipeline: retrieve_document_v2 / search_my_documents_v2 / ... ---
# (backend/mcp_servers/retrieval/langchain_pipeline.py)
#
# Both search legs (pgvector cosine + Postgres full-text) run as SQL, so no
# LangChain retriever classes are needed any more. langchain-core provides
# the Document type; langchain-community the HuggingFace cross-encoder
# wrapper used for reranking (sentence-transformers backs it, model
# downloaded from HF on first use); langchain-openai / langchain-ollama back
# get_langchain_embedding (backend/config/llm_client.py) — OpenAI in PROD,
# Ollama in DEV, and must match the model documents were indexed with.

langchain>=0.3,<0.4
langchain-core>=0.3,<0.4
langchain-community>=0.3,<0.4
langchain-openai>=0.3,<0.4
langchain-ollama>=0.3,<0.4
sentence-transformers>=3.0,<4

# --- web_search / fetch_page tools (backend/mcp_servers/retrieval/web_search.py) ---
#
# ddgs (formerly duckduckgo_search) backs web_search - no API key needed.
